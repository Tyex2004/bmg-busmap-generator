class MainWindow
{
	构造函数()
	{
		加载窗体;
		移除道路预览面板;
		移除线路图面板;
		Program.RPSkiaElement = SkiaCanvas;
	}
	打开地图(按钮被点击)
	{
		弹窗选择地图;
		等待选择地图结果;
		if (有结果)
		{
			移除欢迎面板;
			if (没有道路预览面板) 放入道路预览面板;
			if (没有线路图面板) 放入线路图面板;
			Program.CurrentMap = 选择地图结果;
			Program.CurrentSkiaSVG 清空;
			刷新数据; 生成; 重置视野;
			SkiaCanvas.重绘();
			弹窗提示打开了地图;
		}
	}
	OnPaintSurface(道路预览图被请求绘制)
	{
		if (Program.CurrentMap 不为空)
		{
			canvas = e.Surface.Canvas;
			canvas.清除();
			加载 Svg;
			应用缩放; 应用平移;
			绘制 Svg;
			if (道路编辑模式, 靠近道路节点) { 显示选择框; }
		}
	}
	SkiaCanvas_MouseDown(按下鼠标)
	{
		if (Program.CurrentMap 不为空)
		{
			if (按的是左键)
			{
				记录起点;
				if (Program.IsEditingRoads) // 道路编辑模式
				{
					if (靠近道路节点)
					{
						记录选中道路节点;
						使用道路节点移动工具 = true;
					}
				}
			}
			if (按的是中键) { 准备拖拽屏幕; }
		}
	}
	SkiaCanvas_MouseMove(移动鼠标)
	{
		Program.RPSkiaElement = SkiaCanvas;
		获取鼠标位置;
		if (Program.CurrentMap 不为空)
		{
			SkiaCanvas.重绘();
			if (道路编辑模式)
			{
				if (使用道路节点移动工具)
				{
					记录终点;
				}
			}
			if (拖拽屏幕) { 执行：拖拽屏幕 }
		}
	}
	SkiaCanvas_MouseUp(抬起鼠标)
	{
		if (Program.CurrentMap 不为空)
		{
			if (松开的是左键)
			{
				记录终点;
				if (道路编辑模式)
				{
					if (使用道路节点工具)
					{
						执行：道路节点工具;
						生成;
						退出使用道路节点工具;
					}
				}
			}
			if (松开的是中键) { 退出拖拽屏幕; }
		}
	}
}
